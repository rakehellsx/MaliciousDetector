cmake_minimum_required(VERSION 3.10)
project(MalwareDetectorWeb VERSION 3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# Qt 5.6+ 支持 Win7 32位（使用 Qt 5.6.x + MSVC2015 x86 编译即可）
# 本 CMakeLists 兼容 Qt 5.6 ~ Qt 5.15
find_package(Qt5 5.6 REQUIRED COMPONENTS
    Core
    Gui
    Widgets
    WebEngineWidgets
    WebChannel
)

# 资源文件
set(RESOURCES
    web/resources.qrc
)

# 头文件
set(HEADERS
    web_src/include/MainWindow.h
    web_src/include/AppBridge.h
)

# 源文件
set(SOURCES
    web_src/src/main.cpp
    web_src/src/MainWindow.cpp
    web_src/src/AppBridge.cpp
)

add_executable(MalwareDetectorWeb WIN32
    ${SOURCES}
    ${HEADERS}
    ${RESOURCES}
)

target_include_directories(MalwareDetectorWeb PRIVATE
    web_src/include
)

target_link_libraries(MalwareDetectorWeb PRIVATE
    Qt5::Core
    Qt5::Gui
    Qt5::Widgets
    Qt5::WebEngineWidgets
    Qt5::WebChannel
)

# Windows 特定设置
if(WIN32)
    # 支持 Win7 需要 Qt 5.6.x，并在 MSVC2015 x86 下编译
    target_compile_definitions(MalwareDetectorWeb PRIVATE
        _WIN32_WINNT=0x0601  # Windows 7
    )
endif()
